E2E Step 1: Start infrastructure

Run from the repository root.

docker network inspect soc_shared >/dev/null 2>&1 || docker network create soc_shared

docker compose --env-file notification/.env -f grafana/docker-compose.yml up -d

docker compose --env-file opencti/.env -f opencti/docker-compose.yml up -d opencti-rabbitmq

docker ps --format 'table {{.Names}}\t{{.Status}}\t{{.Ports}}'

docker exec grafana-kafka-broker-1 kafka-topics \
  --bootstrap-server localhost:9092 \
  --create --if-not-exists \
  --topic pa5220.llm_output \
  --partitions 1 \
  --replication-factor 1
