RABBIT_USER=$(grep '^RABBITMQ_DEFAULT_USER=' opencti/.env | cut -d= -f2-)
RABBIT_PASS=$(grep '^RABBITMQ_DEFAULT_PASS=' opencti/.env | cut -d= -f2-)

docker exec opencti-opencti-rabbitmq-1 rabbitmqadmin -u "$RABBIT_USER" -p "$RABBIT_PASS" \
  declare queue name=e2e.alert.high durable=false

docker exec opencti-opencti-rabbitmq-1 rabbitmqadmin -u "$RABBIT_USER" -p "$RABBIT_PASS" \
  declare queue name=e2e.alert.medium durable=false

docker exec opencti-opencti-rabbitmq-1 rabbitmqadmin -u "$RABBIT_USER" -p "$RABBIT_PASS" \
  declare queue name=e2e.alert.low durable=false

docker exec opencti-opencti-rabbitmq-1 rabbitmqadmin -u "$RABBIT_USER" -p "$RABBIT_PASS" \
  declare binding source=soc.alerts destination_type=queue destination=e2e.alert.high routing_key=alert.HIGH

docker exec opencti-opencti-rabbitmq-1 rabbitmqadmin -u "$RABBIT_USER" -p "$RABBIT_PASS" \
